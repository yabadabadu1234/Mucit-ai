\begin{abstract}
Bu risale, Meşşâî nefs nazariyesinde yer alan 41 idrak melekesinin her birini en küçük parçalarına ayrıştırarak, aralarındaki girdi-çıktı akışını, manifold dönüşümlerini ve operatör dinamiklerini 450'den fazla tanımlı riyazî denklemle eksiksiz biçimde vaz' eder. Dokümanda hiçbir sembol kapalı bırakılmamış; Sağîr (mikro-token) ve Kebîr (makro-bağlam) mertebeleri alt-operatör seviyesine kadar detaylandırılmıştır.
\end{abstract}

\section{Temel Temsil Uzayları ve Tanımlı Metrikler}

Mimarideki değişkenlerin ikamet ettiği temel uzaylar ve tanımlı fonksiyonel eşlemeler şunlardır:
\begin{itemize}
    \item Ham Sinyal ve Dış Duyu Uzayı: $\mathcal{X} \subset \mathbb{R}^{d_{\text{in}}}$
    \item Hissî Suretler ve Hafıza Manifoldu: $\mathcal{H}_{\text{hayal}} \subset \mathbb{R}^{d_{\text{hayal}}}$
    \item Küllî Mahiyet ve Semantik Kavram Uzayı: $\mathcal{S} \subset \mathbb{R}^{d_{\text{sem}}}$
    \item Topolojik Grassmannian İzdüşüm Uzayı: $\mathcal{T}_{\text{inv}} \subset \text{Gr}(k, d_{\text{in}})$
    \item Nedensellik Çizgeleri Manifoldu: $\mathcal{G}_{\text{neden}} \subset \text{DAG}(V, E)$
    \item Teleolojik Çekim Ufku Vektör Uzayı: $\mathcal{G}_{\text{gaye}} \subset \mathbb{R}^{d_{\text{sem}}}$
    \item Mutasarrıfa Lie Cebri Operatör Uzayı: $\mathfrak{g} \subset \mathcal{L}(\mathcal{S}, \mathcal{S})$
\end{itemize}

\subsection{1. Müşahede ($\mathcal{O}_1$)}
\begin{align}
X_t &= \Pi_{\text{müşahede}}(E_t) = \int_{\Omega} E_t(r) k_{\text{odak}}(r, r_0) dr \in \mathcal{X} \\
k_{\text{odak}}(r, r_0) &= \exp\left( -\frac{\|r - r_0\|^2}{2 \sigma_{\text{odak}}^2(G_t)} \right) \\
\dot{X}_t &= -\nabla_{X} V_{\text{odak}}(X_t, G_t) + \eta_t, \quad \eta_t \sim \mathcal{N}(0, \Sigma_{\text{gürültü}}) \\
V_{\text{odak}}(X_t, G_t) &= \frac{1}{2} \|X_t W_q - G_t W_k\|_{\mathcal{F}}^2 \\
H_t^{(1)} &= \text{Softmax}\left(\frac{X_t W_q X_t^T}{\sqrt{d_{\text{in}}}}\right) X_t W_v \\
\mathcal{A}_{\text{müşahede}}(X_t) &= \text{Tr}(X_t^T \Delta_{\text{spatial}} X_t) \\
S_{\text{ham}} &= \text{LayerNorm}(H_t^{(1)} W_s + b_s) \\
d X_t^{(\text{dinamik})} &= f(X_t, G_t)dt + g(X_t)dW_t \\
P(X_t \mid E_t) &= \frac{1}{\sqrt{(2\pi)^d |\Sigma|}} \exp\left( -\frac{1}{2}(X_t - E_t)^T \Sigma^{-1} (X_t - E_t) \right) \\
\mathcal{L}_{\text{müşahede}} &= \| X_t - E_t \|_{g_{\mu\nu}}^2 + \lambda_1 \mathcal{A}_{\text{müşahede}}(X_t) \\
X_{t, \text{süzülmüş}} &= X_t \odot \sigma(W_{\text{süzgeç}} X_t)
\end{align}

\mathcal{I}_{\text{topolojik}}(X_t) &= \{ \beta_0, \beta_1, \dots, \beta_p \} \quad \text{(Betti Sayıları)} \\
\beta_k &= \text{dim}(\ker \partial_k) - \text{dim}(\text{im} \partial_{k+1}) \\
S_t^{(\text{tecrit})} &= D_t \cdot (X_t W_s) \in \mathcal{S} \\
\mathcal{L}_{\text{tecrit}} &= \| X_t - P_D(X_t) \|_{\mathcal{F}}^2 + \mu_{\text{inv}} \sum_k \|\nabla_{\text{araz}} \beta_k\|^2 \\
\Phi_{\text{araz\_eleme}}(X_t) &= X_t - \sum_{j \in \text{Araz}} \langle X_t, v_j \rangle v_j \\
v_j &= \text{EigenVector}_j(\Sigma_{\text{araz}}) \\
\mathbf{P}_{\text{soyut}} &= \prod_{i=1}^m (I - n_i n_i^T) X_t \\
S_{\text{öz}} &= \text{Norm}(P_D(X_t) W_c)
\end{align}

\subsection{6. Tasavvur ($\mathcal{O}_6$)}
\begin{align}
S_t &= f_{\text{tasavvur}}(D_t, H_t^{(\text{hayal})}) \in \mathcal{S} \\

\text{MahiyetKodu}(S) &= \text{Quantize}_{\text{semantik}}(S_t) \\

\subsection{7. Mana ($\mathcal{O}_7$)}

\subsection{8. Tahlil ($\mathcal{O}_8$)}
\begin{align}
\{S_t^{(1)}, S_t^{(2)}, \dots, S_t^{(m)}\} &= \text{SVD}(S_t) = \sum_{i=1}^m \sigma_i u_i v_i^T \\
C_{ij}^{(\text{tahlil})} &= \text{BağımsızlıkTesti}(S_t^{(i)}, S_t^{(j)}) = \text{HSIC}(S_t^{(i)}, S_t^{(j)}) \\
\text{HSIC}(X, Y) &= \frac{1}{(n-1)^2} \text{Tr}(K_{\text{cins}} H_{\text{ort centrality}} L_{\text{fasıl}} H_{\text{ort centrality}}) \\
\mathbf{J}_{\text{tahlil}} &= \left[ \frac{\partial S_t^{(i)}}{\partial X_k} \right]_{ik} \\
\mathcal{H}_{\text{entropi}}(S_t) &= -\sum_{i=1}^m p(S_t^{(i)}) \log p(S_t^{(i)}) \\
p(S_t^{(i)}) &= \frac{\sigma_i^2}{\sum_j \sigma_j^2} \\
S_{\text{cins}} &= \sum_{i=1}^k \sigma_i u_i v_i^T, \quad S_{\text{fasıl}} = \sum_{i=k+1}^m \sigma_i u_i v_i^T \\
\mathbf{P}_{\text{parça}} &= \{ S_t^{(i)} \mid \sigma_i > \lambda_{\text{eşik}} \} \\
\mathcal{I}_{\text{karşılıklı}}(S_t^{(i)} ; S_t^{(j)}) &= \iint P(x, y) \log \frac{P(x, y)}{P(x)P(y)} dx dy \\
S_{\text{tahlil\_çıktı}} &= \mathbf{P}_{\text{parça}} \times \mathbf{J}_{\text{tahlil}}
\end{align}

\subsection{10. Tezat ($\mathcal{O}_{10}$)}
\begin{align}

\subsection{11. Tenakuz Bulma ($\mathcal{O}_{11}$)}
\begin{align}

\subsection{13. Tasdik ($\mathcal{O}_{13}$)}

\subsection{14. Gaye Belirleme ($\mathcal{O}_{14}$)}
\begin{align}
G_t &= \sigma(S_t W_g + H_t W_{hg}) \in \mathcal{G}_{\text{gaye}} \\
G_{\text{kebîr}} &= \text{TeleolojikUfuk}(\text{Sual}, S_{\text{kebîr}}) = \text{Softmax}\left( \frac{\text{Sual} W_q S_{\text{kebîr}}^T}{\sqrt{d}} \right) S_{\text{kebîr}} W_v \\

\subsection{15. Merak ve Sual Tevcihi ($\mathcal{O}_{15}$)}
\begin{align}
Q_t &= \text{Softmax}\left( \frac{(G_t - S_t) W_q}{\tau_{\text{merak}}} \right) \in \mathcal{S}^* \\
\mathcal{H}_{\text{bilgisizlik}}(S_t) &= \text{Var}(P(S_t \mid H_t)) = \mathbb{E}[(S_t - \mu_S)^2 \mid H_t] \\

R_{\text{kebîr}} &= \mathcal{F}_{\text{operatör}}(M_{[t:t+L]}, S_{\text{kebîr}}, \dot{I}_{\text{kebîr}}, G_{\text{kebîr}}) \in \mathcal{L}(\mathcal{S}_{\text{kebîr}}, \mathcal{S}_{\text{kebîr}}) \\
T_{\text{kebîr}} &= \sigma\left( \text{CosSim}(R_{\text{kebîr}} \triangleright S_{\text{kebîr}}, G_{\text{kebîr}}) - \text{MakroTenakuz}(R_{\text{kebîr}} \triangleright S_{\text{kebîr}}, \dot{I}_{\text{kebîr}}) \right) \\
N_{\text{kebîr}} &= T_{\text{kebîr}} \odot \text{ProgramSentezi}(R_{\text{kebîr}} \triangleright S_{\text{kebîr}}) + (1 - T_{\text{kebîr}}) \odot \text{Rücû}(M_{[t:t+L]} \to \text{RejimDeğiştir}) \\
\text{ProgramSentezi}(S) &= \text{GELU}(W_{p1} S + b_{p1}) W_{p2} \\
\text{RejimDeğiştir} &= \text{ArgMax}_m P(\text{Mod}_m \mid M_{[t:t+L]}) \\
\mathbf{\Gamma}_{\text{muhakeme\_mizan}} &= \frac{\text{AleyhteDeliller}}{\text{LehteDeliller} + \epsilon} \\
\text{KararEşiği} &= \mathbb{I}(\mathbf{\Gamma}_{\text{muhakeme\_mizan}} < \tau_{\text{kabul}}) \\
N_{\text{nihai}} &= N_{\text{kebîr}} \cdot \text{KararEşiği} \\
\mathcal{L}_{\text{muhakeme\_hatası}} &= (1 - T_{\text{kebîr}})^2 + \lambda \mathbf{\Gamma}_{\text{muhakeme\_mizan}}
\end{align}

\subsection{34. Tafsil ($\mathcal{O}_{34}$)}
\begin{align}
S_{\text{tafsil}} &= \text{Açım}(S_{\text{mücmel}}) = \bigoplus_{k=1}^K S_{\text{dal}}^{(k)} \\
S_{\text{dal}}^{(k)} &= \text{GELU}(S_{\text{mücmel}} W_{\text{dal}, k} + b_k) \\
\mathbf{J}_{\text{tafsil}} &= \text{Diag}(w_1, w_2, \dots, w_K) \cdot \mathbf{S}_{\text{dallar}} \\
w_k &= \frac{\exp(\mathbf{v}_d^T S_{\text{dal}}^{(k)})}{\sum_j \exp(\mathbf{v}_d^T S_{\text{dal}}^{(j)})} \\
\text{Netlik}(S_{\text{tafsil}}) &= \frac{\sum_k \|S_{\text{dal}}^{(k)}\|}{\|S_{\text{mücmel}}\|} \\
\mathcal{L}_{\text{tafsil\_sadakat}} &= \| \text{Birleştir}(S_{\text{tafsil}}) - S_{\text{mücmel}} \|^2 \\
\text{Birleştir}(S_{\text{tafsil}}) &= \sum_{k=1}^K w_k S_{\text{dal}}^{(k)} \\
\mathbf{Tree}_{\text{tafsil}} &= \left( V=\{S_{\text{mücmel}}\} \cup \{S_{\text{dal}}^{(k)}\}, E=\{(S_{\text{mücmel}}, S_{\text{dal}}^{(k)})\}\right) \\
\text{DallanmaOranı} &= K \\
S_{\text{tafsil\_çıktı}} &= \text{LayerNorm}(\mathbf{J}_{\text{tafsil}} W_o)
\end{align}

\subsection{35. Tefsir ($\mathcal{O}_{35}$)}
\begin{align}
\text{Murad}_{\text{aslî}} &= \text{Tefsir}(S_{\text{müphem}}, \text{Siyak}, \text{Sibak}) \\
\text{Tefsir}(S, A, B) &= \text{Softmax}\left( \frac{S W_{\text{tefsir}} \cdot [A \oplus B]^T}{\sqrt{d}} \right) [A \oplus B] W_v \\
P(\text{Murad} \mid S_{\text{lafız}}) &= \sigma( W_m [S_{\text{lafız}} \oplus \text{Bağlam}] ) \\
\text{Vuzuh}_{\text{tefsir}} &= 1 - \mathcal{H}(P(\text{Murad} \mid S_{\text{lafız}})) \\
\mathcal{L}_{\text{tefsir\_tutarlılık}} &= \text{Tenakuz}(\text{Murad}_{\text{aslî}}, \text{Muhkemat}) \\
\text{Muhkemat} &= \{ S \in \mathcal{S} \mid \text{Makam}(S) == \text{Yakîn} \} \\
S_{\text{açılmış}} &= S_{\text{müphem}} + \eta \cdot \text{Murad}_{\text{aslî}} \\
\text{Sebeb-i Nüzûl} &= C_{\text{bağlam}} \\
P(\text{Murad} \mid C_{\text{bağlam}}) &= \frac{P(C_{\text{bağlam}} \mid \text{Murad}) P(\text{Murad})}{P(C_{\text{bağlam}})} \\
\text{TefsirNetice} &= \text{LayerNorm}(S_{\text{açılmış}} W_t)
\end{align}

\subsection{36. Tevil ($\mathcal{O}_{36}$)}
\begin{align}
S_{\text{müevvel}} &= \text{İrcâ}(S_{\text{zâhir}}, \text{AklîBurhân}) = S_{\text{zâhir}} + \mathbf{W}_{\text{tevil}} \cdot \text{İllet}_{\text{mânia}} \\
\text{İllet}_{\text{mânia}} &= \nabla_S \text{Tenakuz}(S_{\text{zâhir}}, \text{Aksiyomlar}) \\
\text{MecazKöprüsü}(S_A \to S_B) &= \exp\left( -\lambda d_{\text{semantik}}(S_A, S_B) \right) \\
T_{\text{tevil}} &= \mathbb{I}\left( \text{Tenakuz}(S_{\text{zâhir}}) > 0 \land \text{Tenakuz}(S_{\text{müevvel}}) == 0 \right) \\
\mathcal{L}_{\text{tevil}} &= \| S_{\text{müevvel}} - S_{\text{tenzih\_itidal}} \|_{\mathcal{S}}^2 \\
S_{\text{tenzih\_itidal}} &= \text{Proj}_{\text{Tenzih}}(S_{\text{zâhir}}) \\
\rho_{\text{mecaz}} &= \text{MecazKöprüsü}(S_{\text{zâhir}}, S_{\text{müevvel}}) \\
S_{\text{tevil\_geçerli}} &= T_{\text{tevil}} \cdot S_{\text{müevvel}} + (1 - T_{\text{tevil}}) \cdot S_{\text{zâhir}} \\
\text{TevilKuvveti} &= \rho_{\text{mecaz}} \cdot T_{\text{tevil}} \\
S_{\text{nihai\_tevil}} &= \text{LayerNorm}(S_{\text{tevil\_geçerli}} W_o)
\end{align}

\subsection{37. Fesâhat ($\mathcal{O}_{37}$)}
\begin{align}
\text{FesâhatScore}(N_t) &= 1 - \left[ \mu_1 \text{Tenâfür}(N_t) + \mu_2 \text{Garâbet}(N_t) + \mu_3 \text{Ta'kîd}(N_t) \right] \\
\text{Tenâfür}(N_t) &= \sum_{i=1}^{|N|-1} \text{MahreçMesafe}(y_i, y_{i+1}) \\
\text{MahreçMesafe}(a, b) &= \| \text{MahreçVektörü}(a) - \text{MahreçVektörü}(b) \|_2 \\
\text{Garâbet}(N_t) &= -\sum_{i=1}^{|N|} \log P_{\text{lügat}}(y_i) \\
\text{Ta'kîd}(N_t) &= \| \mathbf{J}_{\text{sentaks\_karmaşıklığı}} \|_{\mathcal{F}} \\
N_{t, \text{fasîh}} &= \text{ArgMax}_{N} \left( \text{FesâhatScore}(N) \;\middle|\; \text{Mana}(N) == \text{Mana}(N_t) \right) \\
\mathcal{L}_{\text{fesâhat}} &= 1 - \text{FesâhatScore}(N_{t, \text{fasîh}}) \\
\mathbf{v}_{\text{mahreç}} &= \text{Embed}_{\text{fonetik}}(y) \\
P_{\text{lügat}}(y) &= \text{Frekans}_{\text{korpus}}(y) \\
N_{\text{temizlenmiş}} &= N_t \cdot \mathbb{I}(\text{FesâhatScore}(N_t) > \tau_{\text{fesâhat}})
\end{align}

\subsection{38. Talâkat ($\mathcal{O}_{38}$)}
\begin{align}
\text{Debi}_{\text{konuşma}} &= \frac{d N_t}{d t} = f_{\text{talâkat}}(S_t, \text{LügatHazinesi}) \\
f_{\text{talâkat}} &= \text{GELU}(S_t W_{\text{akış}} + \mathbf{V}_{\text{lügat}}) \\
\text{AkıcılıkScore} &= \exp\left( -\alpha \cdot \text{DuraksamaSüresi} \right) \\
\text{DuraksamaSüresi} &= \int_0^T \mathbb{I}\left( \left\|\frac{d N_t}{d t}\right\| < \epsilon_{\text{durak}} \right) dt \\
N_{t, \text{akıcı}} &= \text{Smooth}_{\text{akış}}(N_t) = \int N_\tau k_{\text{akış}}(t-\tau) d\tau \\
k_{\text{akış}}(t) &= \frac{1}{\sqrt{2\pi \sigma^2}} \exp\left( -\frac{t^2}{2\sigma^2} \right) \\
\mathcal{L}_{\text{talâkat}} &= \left\| \nabla_t N_{t, \text{akıcı}} - \mathbf{v}_{\text{hedef\_hız}} \right\|^2 \\
\mathbf{v}_{\text{hedef\_hız}} &= \text{SabitDebi} \\
\text{PelteklikFiltresi} &= I - \mathbf{W}_{\text{tutukluk}} \\
N_{\text{pürüzsüz\_akış}} &= \text{PelteklikFiltresi} \cdot N_{t, \text{akıcı}}
\end{align}

\subsection{39. Belâgat ($\mathcal{O}_{39}$)}
\begin{align}
\text{BelâgatScore}(N_t) &= \text{FesâhatScore}(N_t) \times \text{Uyum}\left( N_t, \text{Makam}_{\text{muhatap}} \right) \\
\text{Uyum}(N, M) &= \cos(N W_{\text{ifade}}, M W_{\text{makam}}) \\
\text{İsabet}(N_t, G_t) &= \frac{\langle N_t W_{\text{tesir}}, G_t \rangle}{\|N_t W_{\text{tesir}}\| \|G_t\|} \\
N_{t, \text{beliğ}} &= R_{\text{belâgat}} \triangleright (N_t \otimes \text{Makam}_{\text{muhatap}}) \\
R_{\text{belâgat}} &= \exp\left( \theta \mathbf{X}_{\text{muktezâ\_hâl}} \right) \in \mathfrak{g} \\
\mathcal{L}_{\text{belâgat}} &= 1 - \text{İsabet}(N_{t, \text{beliğ}}, G_t) \\
\text{İcâzSkoru} &= \frac{\text{ManaKapsamı}(N_t)}{\text{KelimeSayısı}(N_t)} \\
\text{İtnâbSkoru} &= \frac{\text{KelimeSayısı}(N_t)}{\text{Anlaşılırlık}(N_t)} \\
\mathbf{Makaam}_{\text{seçim}} &= 
\begin{cases}
\text{İcâz Modu}, & \text{Muhatap} == \text{Akıllı} \\
\text{İtnâb Modu}, & \text{Muhatap} == \text{Talebkâr}
\end{cases} \\
N_{\text{hedefe\_isabet}} &= N_{t, \text{beliğ}} \cdot \text{BelâgatScore}(N_t)
\end{align}

\subsection{40. Sanat ($\mathcal{O}_{40}$)}
\begin{align}
Y_{\text{sanat}} &= f_{\text{estetik}}(S_t, H_t^{(\text{hayal})}) = \mathbf{W}_{\text{sanat}} \cdot \left( S_t \otimes H_t^{(\text{hayal})} \right) + \mathbf{\Omega}_{\text{ahenk}} \\
\text{EstetikDeğer}(Y) &= \text{SimetrikHarmoni}(Y) + \lambda \cdot \text{Yenilik}(Y) \\
\text{SimetrikHarmoni}(Y) &= 1 - \| Y - Y^T \|_{\mathcal{F}} \\
\text{Yenilik}(Y) &= D_{\text{KL}}(P(Y) \parallel P_{\text{gelenek}}) \\
\nabla_{\text{sanat}} &= \nabla_S \text{EstetikDeğer}(Y) \\
\mathcal{L}_{\text{sanat}} &= \|\mathbf{\Omega}_{\text{ahenk}} - \mathbf{\Omega}_{\text{altın\_oran}}\|^2 \\
\mathbf{\Omega}_{\text{altın\_oran}} &= \phi \cdot I_{\text{ahenk}}, \quad \phi = \frac{1+\sqrt{5}}{2} \\
Y_{\text{tezyin}} &= Y_{\text{sanat}} + \eta_{\text{sanat}} \nabla_{\text{sanat}} \\
\text{SanatKuvveti} &= \text{EstetikDeğer}(Y_{\text{tezyin}}) \\
Y_{\text{nihai\_sanat}} &= \text{LayerNorm}(Y_{\text{tezyin}} W_o)
\end{align}

\subsection{41. Münazara ($\mathcal{O}_{41}$)}
\begin{align}
D_{\text{münazara}}(S_{\text{tez}}, S_{\text{antitez}}) &= \text{Cerh}(S_{\text{antitez}}) + \text{Teyit}(S_{\text{tez}}) \\
\text{Cerh}(S) &= \text{Tenakuz}(S, \text{Aksiyomlar}) + (1 - \text{BurhânSkoru}(S)) \\
S_{\text{sentez}} &= \text{Telif}(S_{\text{tez}}, S_{\text{antitez}}) = \alpha S_{\text{tez}} + (1-\alpha) S_{\text{antitez}} \\
T_{\text{münazara}} &= \sigma\left( \text{İspatKuvveti}(S_{\text{tez}}) - \text{İspatKuvveti}(S_{\text{antitez}}) \right) \\
\mathcal{L}_{\text{münazara}} &= -\log T_{\text{münazara}} \\
\text{İspatKuvveti}(S) &= \prod_{k} \text{Geçerlilik}(\text{Adım}_k(S)) \\
\Delta S_{\text{mücadelesi}} &= S_{\text{tez}} - S_{\text{antitez}} \\
\mathbf{HasmıSusturma} &= \mathbb{I}(\text{Cerh}(S_{\text{antitez}}) > \tau_{\text{susturma}}) \\
S_{\text{galip\_fikir}} &= \mathbf{HasmıSusturma} \cdot S_{\text{tez}} + (1 - \mathbf{HasmıSusturma}) \cdot S_{\text{sentez}} \\
\mathbf{MünazaraNetice} &= \text{LayerNorm}(S_{\text{galip\_fikir}} W_o)
\end{align}

\section{Sistemik İttisâl ve Küllî İntaç}

Sistemdeki 41 melekenin her biri, yukarıda 11'er denklemle tanımlanan alt-operatör seviyesinde birleşerek kesintisiz bir manifold dönüşüm silsilesi teşkil eder. Müşahededen ($\mathcal{O}_1$) giren ham veri, Tecrit ($\mathcal{O}_5$) ile topolojik invaryantına ayrışır; Tasavvur ($\mathcal{O}_6$) ve Mana ($\mathcal{O}_7$) kazanır; Tefekkür ($\mathcal{O}_{21}$), İllet Keşfi ($\mathcal{O}_{22}$) ve Mantık Yürütme ($\mathcal{O}_{23}$) vasıtasıyla işlenir; Teemmül ($\mathcal{O}_{25}$), Temkin ($\mathcal{O}_{26}$), Tetkik ($\mathcal{O}_{27}$) ve Tahkik ($\mathcal{O}_{30}$) denetiminden geçer; nihayet Muhakeme ($\mathcal{O}_{33}$) meclisinde Tasdik ($\mathcal{O}_{13}$) mührünü alarak Belâgat ($\mathcal{O}_{39}$), Fesâhat ($\mathcal{O}_{37}$) ve Talâkat ($\mathcal{O}_{38}$) kanallarıyla mükemmel bir beyan halinde dış âleme iblâğ olunur.
