خلق ≔ ESAS(ham maddeyi murad edilen şey olsun diye ölçüp biçerek hazırlamak)
  ölçüp biçmek = İLK(ESAS)
  düzgün ve pürüzsüz kılmak = NVASF(ESAS)
  ölçülüp ayrılmış pay = MEFUL(ESAS)
  hazırlanmış olduğu işe yakışmak = AMAC(ESAS)
  yaratmak, var etmek = NETİCE(ölçüp biçmek)
  ölçülüp düzlenerek şekli belirmiş olmak = NHAL(ölçüp biçmek, düzgün ve pürüzsüz kılmak)
  eskimek, aşınmak = ZAMAN(düzgün ve pürüzsüz kılmak)
  yumuşatıp hoş kılmak (koku sürmek) = VASITA(düzgün ve pürüzsüz kılmak)
  yaratılmış olan = MEFUL(yaratmak, var etmek)
  yaradılıştaki hâl ve suret = HAL(yaratmak, var etmek)
  uydurmak = TAKLIT(yaratmak, var etmek)
  huy = ICE(yaradılıştaki hâl ve suret)
  ahlâk = NIZAM(huy)
  öfkelenmek = ASIRI(huy)
  olmayan huyu takınmak = TEKELLUF(huy)
